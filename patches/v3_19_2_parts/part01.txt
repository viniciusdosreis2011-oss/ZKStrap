def rep(old, new, label, count=1):
    global s
    if old not in s:
        raise SystemExit(f'v3.19.2 patch: pattern not found: {label}')
    s = s.replace(old, new, count)


def regex_rep(pattern, replacement, label, flags=re.S):
    global s
    ns, n = re.subn(pattern, replacement, s, count=1, flags=flags)
    if n != 1:
        raise SystemExit(f'v3.19.2 patch: regex pattern not found/ambiguous: {label} ({n})')
    s = ns
